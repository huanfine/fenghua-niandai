function calculateClanProfileChecksum(family) {
  if (!family) return '';
  const p = (family.clanTrust && family.clanTrust.prestigeScore) || 0;
  const m = (family.clanTrust && family.clanTrust.trustFundMoney) || 0;
  const g = (family.genealogyList && family.genealogyList.length) || 0;
  const id = family.familyId || '';
  const str = `${id}:${family.surname}:${family.province}:${family.hallName}:${p}:${m}:${g}:${SERVER_CLAN_SALT}`;
  let hash = 0x811c9dc5;
  for (let i = 0; i < str.length; i++) {
    hash ^= str.charCodeAt(i);
    hash = Math.imul(hash, 0x01000193);
  }
  return (hash >>> 0).toString(16);
}

exports.main = async (event, context) => {
  const wxContext = cloud.getWXContext();
  const openid = wxContext.OPENID || 'mock_openid';
  const { action = 'getLeaderboard', seasonId = 'RS_2026_Q4_HANWEI', profile = null, checksum = '' } = event || {};

  const db = cloud.database();

  // 1. 同步家族档案至云端数据库
  if (action === 'syncFamily') {
    if (!profile || !profile.surname || !profile.province || !profile.hallName) {
      return { success: false, reason: 'INVALID_CLAN_DATA' };
    }

    const calculatedSign = calculateClanProfileChecksum(profile);
    if (checksum && checksum !== calculatedSign) {
      return { success: false, reason: 'CHECKSUM_MISMATCH' };
    }

    try {
      // 竞态防御：校验是否已有更早立族的传世宗门
      const existingRes = await db.collection('clans').where({ openid }).get();
      if (existingRes && existingRes.data && existingRes.data.length > 0) {
        const existingClan = existingRes.data[0];
        // 若客户端传入的新档案尝试以不同姓氏/籍贯覆盖已有多代传承的云端旧档案，拦截拒绝！
        const isOverwriteConflict = existingClan.familyId !== profile.familyId &&
          (existingClan.surname !== profile.surname || existingClan.province !== profile.province);
        const existingGenerations = (existingClan.genealogyList && existingClan.genealogyList.length) || 0;
        const newGenerations = (profile.genealogyList && profile.genealogyList.length) || 0;

        if (isOverwriteConflict && existingGenerations > newGenerations) {
          return {
            success: false,
            reason: 'CLAN_ALREADY_EXISTS',
            existingProfile: existingClan,
            message: `该用户已存在传世宗族【${existingClan.fullName}】，拒绝覆盖！`
          };
        }
      }

      const clanDoc = {
        ...profile,
        openid: openid,
        _cloudChecksum: calculatedSign,
        updatedAt: new Date()
      };

      await db.collection('clans').doc(profile.familyId || openid).set({
        data: clanDoc
      });

      return {
        success: true,
        familyId: profile.familyId,
        openid,
        updatedAt: Date.now()
      };
    } catch (err) {
      return { success: false, reason: 'DB_WRITE_ERROR', error: err.message };
    }
  }

  // 2. 拉取当季云端真实排行榜 (按声望值倒序排序)
  if (action === 'getLeaderboard') {
    try {
      const res = await db.collection('clans')
        .where({
          currentSeasonId: seasonId
        })
        .orderBy('clanTrust.prestigeScore', 'desc')
        .limit(100)
        .get();

      let clans = (res && res.data && res.data.length > 0) ? res.data : SEED_SAMPLE_CLANS;
      return {
        success: true,
        seasonId,
        clans,
        isCloudData: (res && res.data && res.data.length > 0)
      };
    } catch (err) {
      return {
        success: true,
        seasonId,
        clans: SEED_SAMPLE_CLANS,
        isCloudData: false,
        fallback: true
      };
    }
  }

  // 3. 跨设备通过 OPENID 迎回宗族档案
  if (action === 'recoverFamily') {
    try {
      const res = await db.collection('clans').where({ openid }).get();
      if (res && res.data && res.data.length > 0) {
        return {
          success: true,
          profile: res.data[0]
        };
      }
      return { success: false, reason: 'NO_PROFILE_FOUND' };
    } catch (err) {
      return { success: false, reason: 'DB_READ_ERROR', error: err.message };
    }
  }

  // 4. 赛季大结卷历史归档入库 (清零重置与名人堂入驻)
  if (action === 'archiveSeason') {
    try {
      const topClansRes = await db.collection('clans')
        .where({ currentSeasonId: seasonId })
        .orderBy('clanTrust.prestigeScore', 'desc')
        .limit(50)
        .get();

      const archiveRecord = {
        seasonId,
        archivedAt: new Date(),
        archivedBy: openid,
        topClans: (topClansRes && topClansRes.data) || []
      };

      await db.collection('clan_leaderboard_archives').add({
        data: archiveRecord
      });

      return {
        success: true,
        archived: true,
        seasonId,
        archivedCount: archiveRecord.topClans.length
      };
    } catch (err) {
      return { success: false, reason: 'ARCHIVE_FAILED', error: err.message };
    }
  }

  // 5. 官方受信任热更包下发通道 (免审核动态内容，微信云端私有防篡改链路)
  if (action === 'getContentManifest') {
    try {
      const manifestRes = await db.collection('content_manifests')
        .orderBy('updatedAt', 'desc')
        .limit(1)
        .get();

      let manifest = (manifestRes && manifestRes.data && manifestRes.data[0]) ? manifestRes.data[0] : null;

          frontierYear: 2035
        };
      }
      return {
        success: true,
        manifest: manifest,
        serverTs: Date.now()
      };
    } catch (err) {
      return { success: false, reason: 'MANIFEST_FETCH_ERROR', error: err.message };
    }
  }

  // 6. 服务端权威联姻核验与 Nonce 防重放记账 (verifyAlliance)
  if (action === 'verifyAlliance') {
    const { invite = null, targetFamilyId = null } = event || {};
    if (!invite || !invite.fromFamId || !invite.nonce || !invite.token) {
      return { success: false, reason: 'MISSING_INVITE_TOKEN', message: '联姻庚帖数据缺失' };
    }

    try {
      // 查验发起方世家在云端是否存在
      const fromFamRes = await db.collection('clans').where({ familyId: invite.fromFamId }).get();
      if (!fromFamRes || !fromFamRes.data || !fromFamRes.data.length) {
        return { success: false, reason: 'SENDER_FAMILY_NOT_FOUND', message: '发起方世家尚未入驻云端百家谱' };
      }

      // 查验 Nonce 幂等
      const nonceRes = await db.collection('redeemed_alliances').where({ nonce: invite.nonce }).get();
      if (nonceRes && nonceRes.data && nonceRes.data.length > 0) {
        return { success: false, reason: 'NONCE_ALREADY_REDEEMED', message: '此道庚帖已被领受，不可重复缔盟' };
      }

      // 记账存证
      await db.collection('redeemed_alliances').add({
        data: {
          nonce: invite.nonce,
          fromFamId: invite.fromFamId,
          redeemedByOpenId: openid,
          targetFamilyId: targetFamilyId || '',
          redeemedAt: new Date()
        }
      });

      return {
        success: true,
        verified: true,
        fromClan: fromFamRes.data[0]
      };
    } catch (err) {
      return { success: false, reason: 'VERIFY_ALLIANCE_ERROR', error: err.message };
    }
  }

  return { success: false, reason: 'UNKNOWN_ACTION' };
};

const SERVER_SALT_LIFESIM = 'LIFESIM_2026_ANTITAMPER_KEY_#8899';

function computeServerChecksum(score, year, age, rep, clean, runSeed) {
  const str = `${score}:${year}:${age}:${rep}:${clean}:${runSeed}:${SERVER_SALT_LIFESIM}`;
  let hash = 0x811c9dc5;
  for (let i = 0; i < str.length; i++) {
    hash ^= str.charCodeAt(i);
    hash = Math.imul(hash, 0x01000193);
  }
  return (hash >>> 0).toString(16);
}

exports.main = async (event, context) => {
  const wxContext = cloud.getWXContext();
  const openid = wxContext.OPENID;

  const {
    score = 0,
    rawScore = 0,
    grade = 'B',
    title = '',
    runSeed = 0,
    year = 2050,
    age = 78,
    rep = 50,
    clean = 50,
    capital = 0,
    delegatedDecisionsCount = 0,
    checksum = '',
    submitTime = Date.now()
  } = event || {};

  // 1. 基础属性硬边界防篡改检查 (防止修改器直接自报 99999)
  if (typeof rep !== 'number' || rep < 0 || rep > 100) {
    return { verified: false, reason: 'INVALID_REP_BOUNDS' };
  }
  if (typeof clean !== 'number' || clean < 0 || clean > 100) {
    return { verified: false, reason: 'INVALID_CLEAN_BOUNDS' };
  }
  if (typeof capital !== 'number' || capital < 0 || capital > 500000000) {
    return { verified: false, reason: 'INVALID_CAPITAL_BOUNDS' };
  }
  if (typeof age !== 'number' || age < 18 || age > 115) {
    return { verified: false, reason: 'INVALID_AGE_BOUNDS' };
  }

  // 2. 评级底分与计分公式服务端精确重算
  const gradeBaseScoreMap = { 'S': 1000, 'A': 750, 'B': 500, 'C': 250 };
  const baseGradeScore = gradeBaseScoreMap[grade] || 500;
  const expectedRawScore = Math.round(
    baseGradeScore +
    rep * 5 +
    clean * 3 +
    Math.min(500, Math.floor(capital / 5000))
  );

  const delegationPenalty = (delegatedDecisionsCount && delegatedDecisionsCount > 0) ? 0.75 : 1.0;
  const expectedTotalScore = Math.round(expectedRawScore * delegationPenalty);

  // 允许浮点四舍五入容差 2 分
  if (Math.abs(score - expectedTotalScore) > 2) {
    return {
      verified: false,
      reason: 'SCORE_FORMULA_MISMATCH',
      reportedScore: score,
      expectedTotalScore
    };
  }

  // 3. 校验客户端签名
  const expectedClientChecksum = computeServerChecksum(score, year, age, rep, clean, runSeed);
  const isLegacyChecksum = (checksum === (((score ^ (year || 2050)) * 31) >>> 0).toString(16));
  const isSecureChecksum = (checksum === expectedClientChecksum);

  if (!isSecureChecksum && !isLegacyChecksum) {
    return {
      verified: false,
      reason: 'CHECKSUM_SIGNATURE_MISMATCH'
    };
  }

  // 4. (可选) 写入云数据库终局归档集合
  try {
    const db = cloud.database();
    await db.collection('user_endgame_records').add({
      data: {
        openid,
        score: expectedTotalScore,
        grade,
        title,
        year,
        age,
        rep,
        clean,
        capital,
        runSeed,
        verified: true,
        isSecureSignature: isSecureChecksum,
        createdAt: new Date()
      }
    });
  } catch (dbErr) {
    // 若数据库集合未建，不阻塞核心认证流程
  }

  return {
    success: true,
    verified: true,
    score: expectedTotalScore,
    grade,
    title,
    openid,
    serverTimestamp: Date.now()
  };
};

  cloud = require('wx-server-sdk');

  // 本地测试/无头仿真环境降级桩
  cloud = {
    getWXContext: () => ({ OPENID: 'local_openid', APPID: 'local_appid' }),

exports.main = async (event, context) => {
  const wxContext = cloud.getWXContext();
  const openid = wxContext.OPENID;
  const appid = wxContext.APPID;

  const {
    offerId = '',
    buyQuantity = 60,
    payRes = {},
    clientOrderId = '',
    requestType = 'verify' // 'verify' | 'restore'
  } = event || {};

  // 1. 基础参数有效性校验
  if (!openid) {
    return {
      success: false,
      verified: false,
      code: 'MISSING_OPENID',
      message: '无法获取有效的微信用户身份凭据'
    };
  }

  // 2. 恢复购买查询模式 (Restore Orders)
  const db = cloud.database();
  const collection = db.collection('midas_orders');

  if (requestType === 'restore') {
    try {
      const existing = await collection.where({
        openid: openid,
        status: 'VERIFIED'
      }).get();

      const orders = (existing && existing.data) || [];
      return {
        success: true,
        verified: true,
        restored: orders.length > 0,
        count: orders.length,
        orders: orders.map(o => ({
          orderId: o.orderId || o._id,
          createdAt: o.createdAt,
          buyQuantity: o.buyQuantity
        }))
      };
    } catch (e) {
      return {
        success: false,
        verified: false,
        code: 'DB_QUERY_FAIL',
        message: '订单记录查询失败',
        error: e.message || String(e)
      };
    }
  }

  // 3. 正常支付核验记账模式
  // 提取订单唯一凭证（支持 payRes 内部透传或客户端随机单号）
  const uniqueOrderId = (payRes && (payRes.orderId || payRes.outTradeNo || payRes.prepay_id)) ||
                        clientOrderId ||
                        `midas_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`;

  try {
    // 3.1 防重放检测：查询该订单号是否已记账
    const checkDup = await collection.where({
      orderId: uniqueOrderId
    }).get();

    if (checkDup && checkDup.data && checkDup.data.length > 0) {
      return {
        success: true,
        verified: true,
        duplicate: true,
        orderId: uniqueOrderId,
        message: '该订单已核验并记录，权益已发放'
      };
    }

    // 3.2 写入订单流水表
    const record = {
      openid,
      appid,
      offerId,
      buyQuantity: Number(buyQuantity) || 60,
      orderId: uniqueOrderId,
      payRes: typeof payRes === 'object' ? payRes : {},
      status: 'VERIFIED',
      createdAt: Date.now(),
      serverTime: new Date().toISOString()
    };

    const addRes = await collection.add({ data: record });

    return {
      success: true,
      verified: true,
      orderId: uniqueOrderId,
      recordId: addRes._id,
      message: '订单核验通过，流水记账成功'
    };
  } catch (err) {
    // 即使数据库写入发生微小异常，也记录日志并回传安全状态，绝不卡死前端
    console.error('[verifyMidasOrder] Database Error:', err);
    return {
      success: true,
      verified: true,
      warning: 'DB_DEGRADE',
      orderId: uniqueOrderId,
      message: '订单本地已核验，数据库流水写入降级'
    };
  }
};

  ? wx.getSharedCanvas()

function parseKVData(kvList) {
  const result = { score: 0, grade: 'B', title: '平凡一生' };
  if (!Array.isArray(kvList)) return result;
  for (let i = 0; i < kvList.length; i++) {
    const item = kvList[i];
    if (item.key === 'era_score') {
      result.score = parseInt(item.value, 10) || 0;
    } else if (item.key === 'era_grade') {
      result.grade = String(item.value || 'B');
    } else if (item.key === 'era_title') {
      result.title = String(item.value || '');
    }
  }
  return result;
}

function fetchAndRenderFriendLeaderboard() {
  if (typeof wx === 'undefined' || typeof wx.getFriendCloudStorage !== 'function') {
    renderEmptyState("当前环境暂不支持好友排行");
    return;
  }

  wx.getFriendCloudStorage({
    keyList: ['era_score', 'era_grade', 'era_title'],
    success: (res) => {
      const rawList = (res && Array.isArray(res.data)) ? res.data : [];
      _friendDataList = rawList.map((friend) => {
        const parsed = parseKVData(friend.KVDataList);
        return {
          nickname: friend.nickname || '同窗挚友',
          avatarUrl: friend.avatarUrl || '',
          score: parsed.score,
          grade: parsed.grade,
          title: parsed.title
        };
      });

      // 按时代履职综合分从高到低排序
      _friendDataList.sort((a, b) => b.score - a.score);

      // 截取前 10 名
      if (_friendDataList.length > 10) {
        _friendDataList = _friendDataList.slice(0, 10);
      }

      renderFriendLeaderboard(_friendDataList);
    },
    fail: (err) => {
      console.warn('[OpenDataContext] 获取好友排行榜失败:', err);
      renderEmptyState("未能加载好友榜单，请稍后刷新");
    }
  });
}

const AUDIO_NOISE_BUFFERS = {};

function getNoiseBuffer(ctxAudio, key, dur, fillFn) {
  if (!ctxAudio) return null;
  const sampleRate = ctxAudio.sampleRate || 44100;
  const cacheKey = key + '_' + sampleRate;
  if (AUDIO_NOISE_BUFFERS[cacheKey]) {
    return AUDIO_NOISE_BUFFERS[cacheKey];
  }
  try {
    const bufferSize = Math.max(64, Math.floor(sampleRate * dur));
    const buf = ctxAudio.createBuffer(1, bufferSize, sampleRate);
    const output = buf.getChannelData(0);
    fillFn(output, bufferSize, sampleRate);
    AUDIO_NOISE_BUFFERS[cacheKey] = buf;
    return buf;
  } catch (e) {
    return null;
  }
}

function _triggerPcmBuffer(ctxAudio, dest, buf, vol, startTime, rateJitter, portamento) {
  if (!ctxAudio || !dest || !buf) return null;
  const t = startTime !== undefined && startTime !== null ? startTime : ctxAudio.currentTime;
  try {
    const src = ctxAudio.createBufferSource();
    src.buffer = buf;
    if (src.playbackRate) {
      const baseRate = 1.0 + (rateJitter ? (Math.random() * 2 - 1) * rateJitter : 0);
      if (portamento) {
        // 中国传统管弦起指气韵微滑音 (天然气口与指尖轻抹，打破电子机械死板感)
        src.playbackRate.setValueAtTime(baseRate * 0.984, t);
        src.playbackRate.linearRampToValueAtTime(baseRate, t + 0.038);
      } else {
        src.playbackRate.setValueAtTime(baseRate, t);
      }
    }
    const gain = ctxAudio.createGain();
    const targetV = Math.max(0.001, Math.min(0.65, vol || 0.16));
    gain.gain.setValueAtTime(0.0001, t);
    gain.gain.linearRampToValueAtTime(targetV, t + 0.005);
    src.connect(gain);
    gain.connect(dest);
    src.start(t);
    return src;
  } catch (e) {
    return null;
  }
}

// 1. 丝桐真声物理声学 (古琴按音/古筝流水/琵琶清音)
// 引入：双阶非对称指腹揉弦 (Flesh Plectrum Model) + 桐木面板声学共鸣峰 (Body Resonances at 180Hz, 320Hz, 540Hz)
function getKarplusStrongBuffer(ctxAudio, freq, dur, character) {
  const f = Math.max(55, Math.min(1760, Math.round(freq * 10) / 10));
  const d = Math.min(dur || 2.0, 3.2);
  const charKey = character || 'guzheng';
  return getNoiseBuffer(ctxAudio, 'acoustic_str_' + charKey + '_' + f + '_' + d.toFixed(2), d, (out, len, sr) => {
    const period = Math.max(8, Math.round(sr / f));
    const ring = new Float32Array(period);
    
    // 指尖轻抚与真丝缠弦真实激励（非机械阶跃脉冲）：
    // 指腹与指甲接触时间约 3~6ms，形成宽圆抛物线形位移，混入细微皮质摩擦泛音
    for (let i = 0; i < period; i++) {
      const phase = i / period;
      const fundamental = Math.sin(Math.PI * phase);
      const secondHarmonic = 0.42 * Math.sin(2 * Math.PI * phase);
      const thirdHarmonic = 0.18 * Math.sin(3 * Math.PI * phase);
      // 指尖皮肉摩擦温润微扰
      const fingerFriction = (Math.sin(i * 19.123 + f) * 43758.5453 % 1) * 0.12;
      ring[i] = (fundamental * 0.65 + secondHarmonic + thirdHarmonic) + fingerFriction;
    }

    // 桐木共鸣箱二阶带通滤波状态 (Body Formants: 220Hz 琴腔暗香, 520Hz 纳音清脆)
    let b1_1 = 0, b1_2 = 0;
    let b2_1 = 0, b2_2 = 0;
    const loss = charKey === 'guqin' ? 0.9968 : (charKey === 'pipa' ? 0.9948 : 0.9978);
    const lowpassBlend = charKey === 'guqin' ? 0.52 : (charKey === 'pipa' ? 0.32 : 0.38);

    let ptr = 0;
    for (let i = 0; i < len; i++) {
      const nextPtr = (ptr + 1) % period;
      const currentSample = ring[ptr];
      // 真实琴弦损耗循环（带阻尼低通扩散，高频泛音先逝，低频基音余音悠长）
      ring[ptr] = loss * ((1 - lowpassBlend) * currentSample + lowpassBlend * ring[nextPtr]);
      ptr = nextPtr;

      // 桐木面板声学共振箱模拟 (Acoustic Soundboard Coupling)
      b1_1 = 0.92 * b1_1 + currentSample * 0.08;
      b1_2 = 0.88 * b1_2 + (currentSample - b1_1) * 0.14;

      b2_1 = 0.78 * b2_1 + currentSample * 0.22;
      b2_2 = 0.84 * b2_2 + (currentSample - b2_1) * 0.18;

      // 传统丝弦起手吸气软包络与余音袅袅自然对数衰减
      const t = i / sr;
      const touchAtt = t < 0.012 ? Math.sin((t / 0.012) * (Math.PI * 0.5)) : 1.0;
      const naturalDecay = Math.exp(-t / (d * (charKey === 'guqin' ? 0.85 : 0.72)));

      // 桐木共振与原弦振动黄金比例耦合输出
      const combined = (currentSample * 0.55 + b1_2 * 0.35 + b2_2 * 0.25) * touchAtt * naturalDecay;
      out[i] = Math.tanh(combined * 1.35) * 0.82;
    }
  });
}

// 2. 竹管气鸣物理声学 (苦竹竹笛/紫竹洞箫/陶土古埙)
// 引入：真实呼吸气口脉冲 (Breath Chiff & Turbulence) + 簧膜共振 (Bamboo Membrane Buzz) + 演奏者自然微弱颤音
function getAcousticWindBuffer(ctxAudio, freq, dur, inst) {
  const f = Math.round(freq * 10) / 10;
  const d = Math.min(dur || 2.2, 4.6);
  const kind = inst || 'flute';
  return getNoiseBuffer(ctxAudio, 'acoustic_wind_' + kind + '_' + f + '_' + d.toFixed(2), d, (out, len, sr) => {
    const isXiao = (kind === 'xiao');
    const isXun = (kind === 'xun');
    let phase = 0;
    let breathTurbulence = 0;
    let memoState = 0;

    for (let i = 0; i < len; i++) {
      const t = i / sr;
      // 演奏者腹式呼吸与指尖运指自然微颤 (5.1Hz 人类情感微颤，前 0.3s 不颤，渐入深情)
      const vibratoOnset = t > 0.32 ? Math.min(1.0, (t - 0.32) / 0.6) : 0;
      const vibratoMod = 0.007 * Math.sin(2 * Math.PI * 5.1 * t) * vibratoOnset;
      const currentPitch = f * (1 + vibratoMod);

      phase += (2 * Math.PI * currentPitch) / sr;

      // 气流吹孔湍流物理建模 (Pinkish Breath Jet Turbulence)
      const rawNoise = (Math.random() * 2 - 1);
      breathTurbulence = 0.82 * breathTurbulence + 0.18 * rawNoise;

      // 吹奏起音气口微爆破 (Attack Chiff)
      const chiffGain = t < 0.06 ? (0.28 * Math.pow(1 - t / 0.06, 1.8)) : 0.035;
      const breathNoise = breathTurbulence * chiffGain;

      // 竹管管腔奇次与偶次谐波共鸣
      const h1 = Math.sin(phase);
      const h2 = (isXun ? 0.06 : (isXiao ? 0.16 : 0.28)) * Math.sin(phase * 2);
      const h3 = (isXun ? 0.03 : (isXiao ? 0.08 : 0.14)) * Math.sin(phase * 3);
      const h4 = (isXun ? 0.01 : (isXiao ? 0.03 : 0.07)) * Math.sin(phase * 4);

      // 笛膜非线性振动 (Bamboo Membrane Resonation): 给竹笛注入灵魂的清亮竹膜脆响
      let membraneBuzz = 0;
      if (!isXiao && !isXun) {
        membraneBuzz = 0.14 * Math.sin(phase * 2.5 + h1 * 0.8);
      }

      const rawAir = (h1 * 0.72 + h2 + h3 + h4 + membraneBuzz + breathNoise);

      // 管身腔体声学吸收滤波 (低通温润化)
      memoState = 0.76 * memoState + 0.24 * rawAir;

      // 呼吸起音包络：管乐器的渐进饱满发音过程
      const attDuration = isXun ? 0.12 : (isXiao ? 0.09 : 0.065);
      const attEnv = t < attDuration ? Math.sin((t / attDuration) * (Math.PI * 0.5)) : 1.0;
      // 乐句收气留白渐退
      const relStart = d - 0.28;
      const relEnv = t > relStart ? Math.exp(-(t - relStart) * 11) : 1.0;

      out[i] = Math.tanh(memoState * 1.22) * attEnv * relEnv * 0.78;
    }
  });
}

// 3. 丝竹二胡物理声学 (马尾弓毛与蟒皮琴筒共鸣峰)
// 引入：松香擦弦滞滑动力学 (Stick-Slip Friction) + 琴筒共鸣箱三峰声学滤波
function getAcousticBowedBuffer(ctxAudio, freq, dur) {
  const f = Math.round(freq * 10) / 10;
  const d = Math.min(dur || 2.4, 4.5);
  return getNoiseBuffer(ctxAudio, 'acoustic_erhu_' + f + '_' + d.toFixed(2), d, (out, len, sr) => {
    let phase = 0;
    let bowRosin = 0;
    let tubeRes1 = 0, tubeRes2 = 0;

    for (let i = 0; i < len; i++) {
      const t = i / sr;
      // 二胡揉弦情感颤音 (Vibrato 5.4Hz，带有传统江南丝竹的运指微滑)
      const vibOnset = t > 0.28 ? Math.min(1.0, (t - 0.28) / 0.5) : 0;
      const vib = 0.0085 * Math.sin(2 * Math.PI * 5.4 * t) * vibOnset;
      phase += (2 * Math.PI * f * (1 + vib)) / sr;

      // 松香马尾擦弦颗粒感 (Rosin Friction Noise)
      bowRosin = 0.80 * bowRosin + 0.20 * (Math.random() * 2 - 1);
      const bowBite = bowRosin * (t < 0.10 ? (0.24 * (1 - t / 0.10)) : 0.04);

      // 非对称真实琴弦锯齿/三角混合波
      const pMod = phase % (2 * Math.PI);
      const bowedWave = (pMod < Math.PI * 0.85) ? (pMod / (Math.PI * 0.85) * 2 - 1) : (1 - (pMod - Math.PI * 0.85) / (Math.PI * 1.15) * 2);

      // 红木琴筒与蟒皮双峰共振腔 (Erhu Tube & Python Skin Formants)
      const excited = bowedWave + bowBite;
      tubeRes1 = 0.74 * tubeRes1 + excited * 0.26;
      tubeRes2 = 0.82 * tubeRes2 + tubeRes1 * 0.22;

      // 运弓软起音与行弓深情延音
      const attEnv = t < 0.09 ? Math.pow(t / 0.09, 1.3) : 1.0;
      const relStart = d - 0.32;
      const relEnv = t > relStart ? Math.exp(-(t - relStart) * 9.5) : 1.0;

      out[i] = Math.tanh((tubeRes2 * 0.82 + Math.sin(phase) * 0.18) * 1.4) * attEnv * relEnv * 0.75;
    }
  });
}

function getAcousticShengBuffer(ctxAudio, freq, dur) {
  const f = Math.round(freq * 10) / 10;
  const d = Math.min(dur || 2.5, 4.8);
  return getNoiseBuffer(ctxAudio, 'acoustic_sheng_' + f + '_' + d.toFixed(2), d, (out, len, sr) => {
    let phase = 0;
    let chamber = 0;
    for (let i = 0; i < len; i++) {
      const t = i / sr;
      phase += (2 * Math.PI * f) / sr;
      // 铜簧振动复合声学谐波
      const reed = Math.sin(phase) * 0.80 + Math.sin(phase * 2) * 0.18 + Math.sin(phase * 3) * 0.06;
      chamber = 0.85 * chamber + 0.15 * reed;
      const att = t < 0.14 ? Math.sin((t / 0.14) * (Math.PI * 0.5)) : 1.0;
      const rel = t > d - 0.32 ? Math.exp(-(t - (d - 0.32)) * 8) : 1.0;
      out[i] = chamber * att * rel * 0.68;
    }
  });
}

// 5. 金石金磬模态共振 (编钟/玉磬/八音盒)
function getMetallicModalBuffer(ctxAudio, baseFreq, dur, material) {
  const f = Math.round(baseFreq * 10) / 10;
  const d = Math.min(dur || 1.2, 2.5);
  const mat = material || 'bronze';
  return getNoiseBuffer(ctxAudio, 'metal_' + mat + '_' + f + '_' + d.toFixed(2), d, (out, len, sr) => {
    const modes = (mat === 'jade')
      ? [
          { r: 1.0, a: 0.66, k: 4.5 },
          { r: 2.0, a: 0.26, k: 9.5 },
          { r: 3.16, a: 0.10, k: 20.0 }
        ]
      : [
          { r: 0.5, a: 0.26, k: 1.9 },
          { r: 1.0, a: 0.62, k: 2.3 },
          { r: 1.0025, a: 0.24, k: 2.6 },
          { r: 1.50, a: 0.26, k: 3.8 },
          { r: 2.00, a: 0.18, k: 5.8 },
          { r: 3.00, a: 0.08, k: 11.0 }
        ];

    for (let i = 0; i < len; i++) {
      const t = i / sr;
      const att = t < 0.0035 ? Math.sin((t / 0.0035) * (Math.PI * 0.5)) : 1.0;
      let sum = 0;
      for (let m = 0; m < modes.length; m++) {
        const md = modes[m];
        sum += md.a * Math.exp(-t * md.k) * Math.sin(2 * Math.PI * (f * md.r) * t);
      }
      out[i] = sum * att * 0.68;
    }
  });
}

// 6. 木音轻响与环境敲击 (紫檀木书案/梆子/算盘珠/粉笔/节拍 - 物理超短微模态，彻底消除电子beep音)
function getTactileClickBuffer(ctxAudio, style) {
  const st = style || 'wood_click';
  const dur = (st === 'wood_click') ? 0.022 : ((st === 'chalk' || st === 'abacus_hi') ? 0.024 : 0.038);
  return getNoiseBuffer(ctxAudio, 'tactile_v3_' + st, dur, (out, len, sr) => {
    let f0 = 320, damp = 520, crisp = 0.28;
    if (st === 'wood_click') { f0 = 320; damp = 520; crisp = 0.28; }
    else if (st === 'abacus_hi') { f0 = 960; damp = 340; crisp = 0.22; }
    else if (st === 'abacus_lo') { f0 = 680; damp = 280; crisp = 0.20; }
    else if (st === 'watch_1') { f0 = 1560; damp = 520; crisp = 0.25; }
    else if (st === 'watch_2') { f0 = 1320; damp = 480; crisp = 0.22; }
    else if (st === 'dice') { f0 = 740; damp = 290; crisp = 0.24; }
    else if (st === 'chalk') { f0 = 640; damp = 360; crisp = 0.18; }
    else if (st === 'wood_block') { f0 = 420; damp = 240; crisp = 0.18; }

    for (let i = 0; i < len; i++) {
      const t = i / sr;
      const att = i < 16 ? (i / 16) : 1.0;
      // 超微指尖物理触击瞬态噪声脉冲 (0.8ms)
      const contactPulse = t < 0.0008 ? (Math.random() * 2 - 1) * Math.sin(Math.PI * (t / 0.0008)) * crisp : 0;
      const m1 = Math.exp(-t * damp) * Math.sin(2 * Math.PI * f0 * t);
      const m2 = Math.exp(-t * damp * 2.2) * Math.sin(2 * Math.PI * (f0 * 2.15) * t) * 0.22;
      out[i] = (contactPulse + m1 * 0.65 + m2) * att * 0.72;
    }
  });
}

// 6.1 寿山石印章温润落案 PCM (厚重实木书案物理低频共鸣与印面纸张贴合瞬态)
function getSealStampBuffer(ctxAudio) {
  return getNoiseBuffer(ctxAudio, 'seal_stamp_v3_pcm', 0.13, (out, len, sr) => {
    let prev = 0;
    for (let i = 0; i < len; i++) {
      const t = i / sr;
      const noise = (Math.random() * 2 - 1) - prev * 0.72;
      prev = noise;
      const slap = t < 0.005 ? noise * Math.exp(-t * 620) * 0.32 : 0;
      const desk1 = Math.exp(-t * 38) * Math.sin(2 * Math.PI * 104 * t) * 0.65;
      const desk2 = Math.exp(-t * 82) * Math.sin(2 * Math.PI * 228 * t) * 0.28;
      const desk3 = Math.exp(-t * 190) * Math.sin(2 * Math.PI * 490 * t) * 0.12;
      out[i] = Math.tanh((slap + desk1 + desk2 + desk3) * 1.15) * 0.72;
    }
  });
}

// 7. 温润粉红环境噪声 (用于微风/细雨/车间蒸汽/教室课本翻动)
function getPinkNoiseBuffer(ctxAudio, dur = 3.0) {
  return getNoiseBuffer(ctxAudio, 'pink_amb_' + dur.toFixed(1), dur, (out, len) => {
    let b0 = 0, b1 = 0, b2 = 0, b3 = 0, b4 = 0, b5 = 0, b6 = 0;
    for (let i = 0; i < len; i++) {
      const white = Math.random() * 2 - 1;
      b0 = 0.99886 * b0 + white * 0.0555179;
      b1 = 0.99332 * b1 + white * 0.0750759;
      b2 = 0.96900 * b2 + white * 0.1538520;
      b3 = 0.86650 * b3 + white * 0.3104856;
      b4 = 0.55000 * b4 + white * 0.5329522;
      b5 = -0.7616 * b5 - white * 0.0168980;
      out[i] = (b0 + b1 + b2 + b3 + b4 + b5 + b6 + white * 0.5362) * 0.045;
      b6 = white * 0.115926;
    }
  });
}

// 8. 宣纸书页微翻与书斋案头纸响 (细腻微动)
function getPaperFlipBuffer(ctxAudio, isScroll) {
  const dur = isScroll ? 0.15 : 0.082;
  return getNoiseBuffer(ctxAudio, 'paper_velvet_' + (isScroll ? 'long' : 'short'), dur, (out, len, sr) => {
    let prev1 = 0, prev2 = 0, lp = 0;
    for (let i = 0; i < len; i++) {
      const t = i / sr;
      const normT = t / dur;
      const edgeSnap = t < 0.010 ? Math.sin(Math.PI * (t / 0.010)) * 0.24 : 0;
      const arcEnv = Math.pow(Math.sin(Math.PI * normT), 1.6) * Math.exp(-normT * 1.3);
      const density = 0.13 + 0.16 * Math.sin(Math.PI * normT);
      const r = Math.random();
      const impulse = r < density ? (r < density * 0.5 ? 1 : -1) : 0;
      const diff = impulse - 1.35 * prev1 + 0.45 * prev2;
      prev2 = prev1;
      prev1 = impulse;
      lp = 0.66 * lp + 0.34 * diff;
      out[i] = (edgeSnap * diff * 0.3 + lp * arcEnv * 0.58);
    }
  });
}

// 9. 紫砂茶汤注水与清泉水滴声学 Minnaert 气泡微共鸣
function getWaterPourBuffer(ctxAudio) {
  return getNoiseBuffer(ctxAudio, 'water_pour_minnaert', 0.42, (out, len, sr) => {
    const drops = [
      { t0: 0.010, f0: 620, tau: 0.0070, amp: 0.34 },
      { t0: 0.068, f0: 840, tau: 0.0060, amp: 0.40 },
      { t0: 0.132, f0: 720, tau: 0.0065, amp: 0.34 },
      { t0: 0.198, f0: 960, tau: 0.0055, amp: 0.42 },
      { t0: 0.268, f0: 790, tau: 0.0062, amp: 0.36 },
      { t0: 0.338, f0: 1060, tau: 0.0052, amp: 0.30 }
    ];
    for (let i = 0; i < len; i++) {
      const t = i / sr;
      let s = 0;
      for (let d = 0; d < drops.length; d++) {
        const dr = drops[d];
        const dt = t - dr.t0;
        if (dt >= 0 && dt < 0.038) {
          const instPhase = 2 * Math.PI * dr.f0 * dt * (1 + 0.14 * (dt / 0.028));
          s += dr.amp * Math.exp(-dt / dr.tau) * Math.sin(instPhase);
        }
      }
      out[i] = s * 0.62;
    }
  });
}

function getEffectiveAudioContext() {
  if (_audioCtx) return _audioCtx;
  if (typeof wx !== 'undefined' && typeof wx.createWebAudioContext === 'function') {
    try { _audioCtx = wx.createWebAudioContext(); } catch (e) {}
  }
  if (!_audioCtx && typeof AudioContext !== 'undefined') {
    try { _audioCtx = new AudioContext(); } catch (e) {}
  } else if (!_audioCtx && typeof webkitAudioContext !== 'undefined') {
    try { _audioCtx = new webkitAudioContext(); } catch (e) {}
  }
  if (_audioCtx && _audioCtx.state === 'suspended') {
    try { _audioCtx.resume(); } catch (e) {}
  }
  return _audioCtx;
}

function getMasterAudioBus(ctxAudio) {
  if (!ctxAudio) return null;
  if (_masterBusNode && _masterBusNode._ctx === ctxAudio) return _masterBusNode;
  try {
    const t = ctxAudio.currentTime || 0;
    const masterGain = ctxAudio.createGain();
    masterGain.gain.setValueAtTime(0.80, t);
    masterGain._ctx = ctxAudio;

    const warmthFilter = ctxAudio.createBiquadFilter();
    warmthFilter.type = 'lowpass';
    warmthFilter.frequency.setValueAtTime(6400, t);
    warmthFilter.Q.setValueAtTime(0.5, t);
    masterGain.connect(warmthFilter);

    const dryGain = ctxAudio.createGain();
    dryGain.gain.setValueAtTime(0.78, t);
    warmthFilter.connect(dryGain);

    let lastNode = dryGain;
    if (ctxAudio.createDynamicsCompressor) {
      try {
        const comp = ctxAudio.createDynamicsCompressor();
        comp.threshold.setValueAtTime(-16, t);
        comp.knee.setValueAtTime(20, t);
        comp.ratio.setValueAtTime(2.6, t);
        comp.attack.setValueAtTime(0.012, t);
        comp.release.setValueAtTime(0.20, t);
        lastNode.connect(comp);
        lastNode = comp;
      } catch (e) {}
    }
    lastNode.connect(ctxAudio.destination);
    _masterBusNode = masterGain;
    return _masterBusNode;
  } catch (e) {
    return ctxAudio.destination;
  }
}

function getBgmSubBus(ctxAudio) {
  const master = getMasterAudioBus(ctxAudio);
  if (!ctxAudio || !master) return null;
  if (_bgmBusNode && _bgmBusNode._ctx === ctxAudio) return _bgmBusNode;
  try {
    const t = ctxAudio.currentTime || 0;
    const bus = ctxAudio.createGain();
    bus.gain.setValueAtTime(0.85, t);
    bus._ctx = ctxAudio;

    // 1. 直达声通道 (Dry Path)
    const dryGain = ctxAudio.createGain();
    dryGain.gain.setValueAtTime(0.80, t);
    bus.connect(dryGain);
    dryGain.connect(master);

    // 2. 东方意境殿堂空间漫反射声场 (Acoustic Spatial Ambience & 3-Tap Diffuse Network)
    if (typeof ctxAudio.createDelay === 'function') {
      try {
        const reverbFilter = ctxAudio.createBiquadFilter();
        reverbFilter.type = 'lowpass';
        reverbFilter.frequency.setValueAtTime(2800, t); // 丝绸温润吸音
        reverbFilter.Q.setValueAtTime(0.5, t);

        const delay1 = ctxAudio.createDelay(1.5);
        delay1.delayTime.setValueAtTime(0.068, t); // 早期泛音反射 68ms (左侧侧墙反射)

        const delay2 = ctxAudio.createDelay(1.5);
        delay2.delayTime.setValueAtTime(0.142, t); // 次级空灵扩散 142ms (右侧后墙反射)

        const delay3 = ctxAudio.createDelay(1.5);
        delay3.delayTime.setValueAtTime(0.235, t); // 殿堂穹顶远景扩散 235ms (回字形深邃声场)

        const fb1 = ctxAudio.createGain();
        fb1.gain.setValueAtTime(0.24, t);

        const fb2 = ctxAudio.createGain();
        fb2.gain.setValueAtTime(0.20, t);

        const fb3 = ctxAudio.createGain();
        fb3.gain.setValueAtTime(0.16, t);

        const wetGain = ctxAudio.createGain();
        wetGain.gain.setValueAtTime(0.42, t); // 醇厚饱满、余韵悠长的空间沉浸感

        bus.connect(reverbFilter);
        reverbFilter.connect(delay1);
        delay1.connect(fb1);
        fb1.connect(delay2);
        delay2.connect(fb2);
        fb2.connect(delay3);
        delay3.connect(fb3);
        fb3.connect(delay1);

        delay1.connect(wetGain);
        delay2.connect(wetGain);
        delay3.connect(wetGain);
        wetGain.connect(master);
      } catch (_) {}
    }

    _bgmBusNode = bus;
    return bus;
  } catch (e) {
    return master;
  }
}

function duckBgmForSfx(duckDuration = 1.0, dipLevel = 0.38) {
  const ctxAudio = getEffectiveAudioContext();
  if (!ctxAudio || !_bgmPlaying) return;
  const bus = getBgmSubBus(ctxAudio);
  if (!bus || !bus.gain) return;
  try {
    const t = ctxAudio.currentTime;
    bus.gain.cancelScheduledValues(t);
    bus.gain.setValueAtTime(Math.max(0.001, bus.gain.value || 0.85), t);
    bus.gain.linearRampToValueAtTime(dipLevel, t + 0.045);
    bus.gain.exponentialRampToValueAtTime(0.85, t + Math.max(0.4, duckDuration));
  } catch (e) {}
}

function playInstrumentTone(instrument, freq, duration, volume, delaySec = 0, customBus = null, exactStartTime = null) {
  const ctxAudio = getEffectiveAudioContext();
  if (!ctxAudio || !freq) return;
  try {
    const dest = customBus || getMasterAudioBus(ctxAudio);
    if (!dest) return;
    const baseT = (exactStartTime !== null && exactStartTime !== undefined) ? exactStartTime : ctxAudio.currentTime;
    const t = baseT + (delaySec || 0);
    const v = Math.max(0.005, Math.min(0.28, (volume || 0.08) * 0.95));
    const d = Math.max(0.25, Math.min(duration || 1.4, 4.2));

    if (instrument === 'guqin') {
      const ksBuf = getKarplusStrongBuffer(ctxAudio, freq, Math.min(d, 2.8), 'guqin');
      _triggerPcmBuffer(ctxAudio, dest, ksBuf, v * 0.95, t, 0.001, true);
    } else if (instrument === 'guzheng' || instrument === 'pipa') {
      const isPipa = (instrument === 'pipa');
      const ksBuf = getKarplusStrongBuffer(ctxAudio, freq, Math.min(d, 2.2), isPipa ? 'pipa' : 'guzheng');
      _triggerPcmBuffer(ctxAudio, dest, ksBuf, v * (isPipa ? 0.88 : 0.95), t, 0.001, false);
    } else if (instrument === 'flute' || instrument === 'xiao' || instrument === 'xun') {
      const windBuf = getAcousticWindBuffer(ctxAudio, freq, d, instrument);
      _triggerPcmBuffer(ctxAudio, dest, windBuf, v * 0.88, t, 0.001, true);
    } else if (instrument === 'erhu') {
      const erhuBuf = getAcousticBowedBuffer(ctxAudio, freq, d);
      _triggerPcmBuffer(ctxAudio, dest, erhuBuf, v * 0.86, t, 0.001, true);
    } else if (instrument === 'bianzhong' || instrument === 'bianqing') {
      const mat = instrument === 'bianqing' ? 'jade' : 'bronze';
      const buf = getMetallicModalBuffer(ctxAudio, freq, Math.max(1.2, d), mat);
      _triggerPcmBuffer(ctxAudio, dest, buf, v * 0.85, t, 0.001, false);
    } else if (instrument === 'sheng' || instrument === 'cello_drone' || instrument === 'pad') {
      const shengBuf = getAcousticShengBuffer(ctxAudio, freq, d);
      _triggerPcmBuffer(ctxAudio, dest, shengBuf, v * 0.72, t, 0.001, false);
    } else if (instrument === 'musicbox') {
      const buf = getMetallicModalBuffer(ctxAudio, freq, Math.min(1.4, d), 'jade');
      _triggerPcmBuffer(ctxAudio, dest, buf, v * 0.68, t, 0.001);
    } else if (instrument === 'drum') {
      if (freq < 200) {
        const osc = ctxAudio.createOscillator();
        const dG = ctxAudio.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(92, t);
        osc.frequency.exponentialRampToValueAtTime(46, t + 0.18);
        dG.gain.setValueAtTime(0.0001, t);
        dG.gain.linearRampToValueAtTime(v * 0.55, t + 0.010);
        dG.gain.exponentialRampToValueAtTime(0.0001, t + 0.48);
        osc.connect(dG);
        dG.connect(dest);
        osc.start(t);
        osc.stop(t + 0.50);
      } else {
        const buf = getTactileClickBuffer(ctxAudio, 'wood_block');
        _triggerPcmBuffer(ctxAudio, dest, buf, v * 0.55, t, 0.010);
      }
    } else {
      const buf = getKarplusStrongBuffer(ctxAudio, freq, d, 'guzheng');
      _triggerPcmBuffer(ctxAudio, dest, buf, v * 0.85, t, 0.001);
    }
  } catch (e) {
    _logErrorFn('playInstrumentTone', e);
  }
}

function getCurrentBgmSuite() {
  const state = _getGameStateFn() || {};
  const age = state.age !== undefined ? state.age : 18;
  const yr = state.year ? state.year : 1992;
  if (age < 16) return ERA_BGM_SUITES.childhood;
  if (yr < 2000) return ERA_BGM_SUITES.early;
  if (yr < 2013) return ERA_BGM_SUITES.mid;
  return ERA_BGM_SUITES.modern;
}

function scheduleBgmBeat(exactTime, beatIdx, suite) {
  const ctxAudio = getEffectiveAudioContext();
  if (!ctxAudio) return;
  const bgmBus = getBgmSubBus(ctxAudio);
  if (!bgmBus) return;

  const stepInPhrase = beatIdx % suite.totalBeats;
  const barIdx = Math.floor(stepInPhrase / 4);
  const beatInBar = stepInPhrase % 4;
  const chord = suite.chords[barIdx % suite.chords.length] || suite.chords[0];

  // 计算曲式阶段起伏权重 (起·承·转·合 Dynamic Macro-Contour)
  // bar 0-3 (起: 悠远清幽 0.85), bar 4-7 (承: 渐进展开 0.98), bar 8-11 (转·高潮: 激荡豪迈 1.15), bar 12-15 (合: 沉淀回味 0.88)
  const isIntro = (barIdx < 4);
  const isAscent = (barIdx >= 4 && barIdx < 8);
  const isClimax = (barIdx >= 8 && barIdx < 12);
  const isCoda = (barIdx >= 12);
  const sectionGain = isClimax ? 1.15 : (isAscent ? 0.98 : (isCoda ? 0.88 : 0.85));

  // 人性化微量时间抖动 (Human micro-rubato timing jitter, ±5ms)
  const humanJitter = (Math.sin(stepInPhrase * 3.7) * 0.005);
  const humanTime = exactTime + humanJitter;

  // ==========================================
  // 声部一：溯源基底（远景 / 呼吸低音长鸣，Ambient Sustained Foundation）
  // ==========================================
  if (beatInBar === 0) {
    const droneVol = (isClimax ? 0.038 : (isIntro ? 0.024 : 0.030)) * sectionGain;
    playInstrumentTone('cello_drone', chord[0], suite.beatSec * 3.7, droneVol, 0, bgmBus, humanTime);

    // 和声层：展开与高潮段加入五度与八度笙簧和鸣
    if (isAscent || isClimax) {
      if (chord[1]) playInstrumentTone('pad', chord[1], suite.beatSec * 3.4, droneVol * 0.70, 0.02, bgmBus, humanTime);
      if (chord[2] && isClimax) playInstrumentTone('sheng', chord[2], suite.beatSec * 3.0, droneVol * 0.60, 0.04, bgmBus, humanTime);
    }

    // 结构性段落标志金石之声 (Structural Cadential Accents)
    if (barIdx === 0 || barIdx === 4 || barIdx === 8 || barIdx === 12) {
      if (suite === ERA_BGM_SUITES.modern) {
        // 当代沧海回望：灵璧玉磬悠长空灵微鸣
        playInstrumentTone('bianqing', chord[0] * 4.0, suite.beatSec * 3.0, 0.032, 0.01, bgmBus, humanTime);
      } else if (suite === ERA_BGM_SUITES.mid) {
        // 千禧盛世：曾侯乙青铜编钟宏大鸣响
        playInstrumentTone('bianzhong', chord[0] * 2.0, suite.beatSec * 3.5, 0.036, 0.01, bgmBus, humanTime);
      } else if (suite === ERA_BGM_SUITES.early && isClimax) {
        // 改革春潮高潮：牛皮战鼓沉稳大音
        playInstrumentTone('drum', 85, 0.45, 0.028, 0.01, bgmBus, humanTime);
      }
    }
  }

  // ==========================================
  // 声部二：丝桐流觞（中景 / 琵琶古筝流水织体与和弦琶音，Plucked String Fluid Texture）
  // ==========================================
  // 采用真正的五声调式行云流水织体，区分强弱拍、主辅交替与声部对位
  const arpInst = (suite === ERA_BGM_SUITES.childhood) ? 'musicbox' : ((suite === ERA_BGM_SUITES.early && isClimax) ? 'pipa' : 'guzheng');

  if (beatInBar === 0) {
    // 拍 1：低音抚弦 (Root Pluck) 沉稳有力
    const rootP = chord[0] * 2.0;
    playInstrumentTone(arpInst, rootP, suite.beatSec * 1.8, 0.032 * sectionGain, 0.008, bgmBus, humanTime);
  } else if (beatInBar === 1) {
    // 拍 2：次强拍中音泛音流动
    const fifthP = (chord[1] || chord[0] * 1.5) * 2.0;
    playInstrumentTone(arpInst, fifthP, suite.beatSec * 1.4, 0.024 * sectionGain, 0.012, bgmBus, humanTime);
  } else if (beatInBar === 2) {
    // 拍 3：三度和音回旋，与主旋律交错
    const midP = (chord[2] || chord[0] * 1.25) * 2.0;
    playInstrumentTone(arpInst, midP, suite.beatSec * 1.5, 0.028 * sectionGain, 0.006, bgmBus, humanTime);
  } else if (beatInBar === 3) {
    // 拍 4：高八度水滴泛音收束
    const highP = (chord[3] || chord[0] * 2.0) * 2.0;
    playInstrumentTone(arpInst, highP, suite.beatSec * 1.2, 0.022 * sectionGain, 0.015, bgmBus, humanTime);
  }

  // ==========================================
  // 声部三：灵韵长歌（前景 / 竹笛洞箫二胡主旋律，Expressive Lead Melodic Voice）
  // ==========================================
  let isLeadNotePlayingThisBeat = false;
  if (suite.notes && suite.notes.length) {
    for (let i = 0; i < suite.notes.length; i++) {
      const note = suite.notes[i];
      if (Math.abs(note.beat - stepInPhrase) < 0.01) {
        isLeadNotePlayingThisBeat = true;
        // 随曲式宏观动态起伏智能微调力度 (增加饱满度 0.065~0.088)
        const leadVol = Math.max(0.045, note.vol * 1.28 * sectionGain);
        playInstrumentTone(
          note.inst,
          note.freq,
          note.dur * suite.beatSec,
          leadVol,
          0,
          bgmBus,
          exactTime
        );
        // 在高潮与展开段落加入温润八度/五度和声复调辅奏 (Polyphonic Counterpart)
        if ((isAscent || isClimax) && note.dur >= 1.5) {
          const subInst = (note.inst === 'flute' || note.inst === 'xiao') ? 'erhu' : 'xiao';
          const subFreq = (note.freq > 400) ? note.freq * 0.5 : note.freq * 1.5;
          playInstrumentTone(subInst, subFreq, note.dur * suite.beatSec * 0.85, leadVol * 0.45, 0.03, bgmBus, humanTime);
        }
      }
    }
  }

  // ==========================================
  // 声部四：对位呼应与留白点睛（Call-and-Response & Fill Motif）
  // ==========================================

function getHarmonizedFreq(degreeIdx = 4, octaveMul = 1.0) {
  const scale = getCurrentScale();

function detectCurrentEraScene(evtObj) {

function bgmSchedulerLoop() {
  if (!_bgmPlaying || !_bgmEnabled) return;
  const ctxAudio = getEffectiveAudioContext();
  if (!ctxAudio) return;

  const suite = getCurrentBgmSuite();
  const lookaheadSec = 0.35;

  while (_bgmNextNoteTime < ctxAudio.currentTime + lookaheadSec) {
    scheduleBgmBeat(_bgmNextNoteTime, _bgmBeatCount, suite);
    _bgmNextNoteTime += suite.beatSec;
    _bgmBeatCount++;
  }
}

// 广告位独立单例与事件句柄隔离缓存
let _cachedInspirationAd = null;
let _cachedInspirationHandler = null;
let _lastInspirationAdTs = 0;

let _cachedEmergencyAd = null;
let _cachedEmergencyHandler = null;

let _cachedFateAd = null;
let _cachedFateHandler = null;

function createMonetizationManager(deps = {}) {
  const {
    getState = () => ({}),
    getConfig = () => ({}),
    showToast = () => {},
    reportMilestone = () => {},
    saveGameState = () => {},
    requestRender = () => {},

  } = deps;

  // 1. 终局家风传承激励视频广告
  function watchRewardedVideoForInspiration(onSuccess) {
    const state = getState();
    const config = getConfig().monetization || {};
    const rewardsCfg = getConfig().rewards || {};
    const maxPerRun = config.maxAdTalentsPerRun || 1;
    const maxTotal = config.maxTotalHeritageTalents || 5;
    const cooldownMs = config.adCooldownMs || 60000;
    const adUnitId = config.inspirationAdUnitId || config.adUnitId || '';

    if ((state.rewardedAdWatchedThisRun || 0) >= maxPerRun) {
      showToast("📽️ 这一世的旧影已翻阅过了，留待后人再看吧", "info");
      return;
    }
    if ((state.heritageBonusTalents || 0) >= maxTotal) {
      showToast("📽️ 家学底蕴已攒足五分，不必再添了", "gold");
      return;
    }
    const nowTs = Date.now();
    if (_lastInspirationAdTs > 0 && nowTs - _lastInspirationAdTs < cooldownMs) {
      const waitSec = Math.ceil((cooldownMs - (nowTs - _lastInspirationAdTs)) / 1000);
      showToast(`📽️ 往事尚在回味中，稍候 ${waitSec} 秒再翻阅吧`, "info");
      return;
    }

    reportMilestone('ad_click', {
      adType: 'endgame_heritage',
      abBucket: state.abTestBucket || 'A_standard'
    });

    const grantAdInspiration = () => {
      if ((state.rewardedAdWatchedThisRun || 0) >= maxPerRun) return;
      state.rewardedAdWatchedThisRun = (state.rewardedAdWatchedThisRun || 0) + 1;
      state.heritageBonusTalents = Math.min(maxTotal, (state.heritageBonusTalents || 0) + 1);
      const isBucketB = (state.abTestBucket === 'B_enhanced');
      if (isBucketB) {
        state.heritageBonusCapital = (state.heritageBonusCapital || 0) + (rewardsCfg.heritageBundleCash || 15000);
        state.heritageBonusRep = (state.heritageBonusRep || 0) + (rewardsCfg.heritageBundleRep || 5);
        state.hasHeirloomSeal = true;
      }
      _lastInspirationAdTs = Date.now();
      saveGameState(true);
      reportMilestone('ad_completed', {
        adType: 'endgame_heritage',
        abBucket: state.abTestBucket || 'A_standard'
      });
      if (typeof onSuccess === 'function') {
        onSuccess(true);
      } else {
        if (isBucketB) {
          showToast("📽️ 参悟先祖遗韵！下代天赋+1、获¥15,000祖荫银两与世家声誉！", "gold");
        } else {
          showToast("📽️ 翻阅旧影有所得，为后人留下一分家学底蕴（下局天赋+1）", "gold");
        }
      }
      requestRender();
    };

    if (isPlaceholderOrEmpty(adUnitId)) {
      reportMilestone('ad_failed', {
        adType: 'endgame_heritage',
        reason: 'unconfigured_or_placeholder_adunit',
        abBucket: state.abTestBucket || 'A_standard'
      });
      console.warn('[LifeSim:Ad] 传承激励视频广告位ID尚未在 gameConfig.js 中配置，跳过拉起广告组件。');
      showToast("📽️ 时代影像正在筹备中（广告位未配置）", "info");
      return;
    }

    if (typeof wx !== 'undefined' && typeof wx.createRewardedVideoAd === 'function') {
      if (!_cachedInspirationAd) {
        _cachedInspirationAd = wx.createRewardedVideoAd({ adUnitId: adUnitId });
      }
      const videoAd = _cachedInspirationAd;
      if (videoAd) {
        if (_cachedInspirationHandler && typeof videoAd.offClose === 'function') {
          videoAd.offClose(_cachedInspirationHandler);
        }
        _cachedInspirationHandler = (res) => {
          if (res && res.isEnded === true) {
            grantAdInspiration();
          } else {
            reportMilestone('ad_skipped', {
              adType: 'endgame_heritage',
              abBucket: state.abTestBucket || 'A_standard'
            });
            showToast("📽️ 需完整观阅旧影方可领悟家学点化", "info");
            if (typeof onSuccess === 'function') onSuccess(false);
          }
        };
        if (typeof videoAd.onClose === 'function') {
          videoAd.onClose(_cachedInspirationHandler);
        }
        if (typeof videoAd.show === 'function') {
          videoAd.show().catch(() => {
            if (typeof videoAd.load === 'function') {
              videoAd.load().then(() => videoAd.show()).catch((e) => {
                logDebugError('inspirationAd.show', e);
                reportMilestone('ad_failed', {
                  adType: 'endgame_heritage',
                  reason: (e && e.errMsg) || 'load_fail',
                  abBucket: state.abTestBucket || 'A_standard'
                });
                showToast("📽️ 旧影载入未竟，请稍候再试", "info");
                if (typeof onSuccess === 'function') onSuccess(false);
              });
            } else {
              reportMilestone('ad_failed', {
                adType: 'endgame_heritage',
                reason: 'show_fail',
                abBucket: state.abTestBucket || 'A_standard'
              });
              showToast("📽️ 旧影载入未竟，请稍候再试", "info");
              if (typeof onSuccess === 'function') onSuccess(false);
            }
          });
          return;
        }
      }
    }
    reportMilestone('ad_failed', {
      adType: 'endgame_heritage',
      reason: 'no_ad_api',
      abBucket: state.abTestBucket || 'A_standard'
    });
    showToast("📽️ 暂无可供翻阅的旧影，请稍候再试", "info");
    if (typeof onSuccess === 'function') onSuccess(false);
  }

  function canUseEmergencyMedical() {
    const state = getState();
    const config = getConfig().monetization || {};
    const maxEmergencyPerRun = config.maxEmergencyMedicalPerRun || 1;
    if ((state.emergencyMedicalUsedCount || 0) >= maxEmergencyPerRun) return false;
    if (state.emergencyMedicalCooldownUntil && (state.year || 1990) < state.emergencyMedicalCooldownUntil) return false;
    return true;
  }

        }
      });
      return;
    }

    // 模式 B：企业主体 / 有版号虚拟支付模式 (Midas Payment)
    // 1. iOS 端受苹果 IAP 政策约束，微信平台不支持直接拉起米大师 Android 支付，走合规短片赠予
    if (sysPlatform === 'ios' || sysPlatform === 'iphone' || sysPlatform === 'ipad') {

        }
      });
      return;
    }

    // 2. 检查 OfferId 是否有效
    if (isPlaceholderOrEmpty(config.midasOfferId)) {
      console.warn('[LifeSim:Midas] 米大师 OfferId 未配置。');
      if (sysPlatform === 'devtools') {
        grantSponsorEditionRewards();

      }
      return;
    }

    // 3. 正常拉起微信米大师虚拟支付
    if (typeof wx !== 'undefined' && typeof wx.requestMidasPayment === 'function') {
      const isDevEnv = sysPlatform === 'devtools' ? 1 : 0;
      const buyQuantity = config.midasBuyQuantity || 60;
      const clientOrderId = `midas_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`;

      wx.requestMidasPayment({

/**
 * 计算家族档案防篡改哈希签名
 */
function calculateClanProfileChecksum(family) {
  if (!family) return '';
  const p = (family.clanTrust && family.clanTrust.prestigeScore) || 0;
  const m = (family.clanTrust && family.clanTrust.trustFundMoney) || 0;
  const g = (family.genealogyList && family.genealogyList.length) || 0;
  const id = family.familyId || '';
  const str = `${id}:${family.surname}:${family.province}:${family.hallName}:${p}:${m}:${g}:${SERVER_CLAN_SALT}`;
  let hash = 0x811c9dc5;
  for (let i = 0; i < str.length; i++) {
    hash ^= str.charCodeAt(i);
    hash = Math.imul(hash, 0x01000193);
  }
  return (hash >>> 0).toString(16);
}

/**
 * 4. 真实全服百家榜单动态汇聚与现实赛季定期归档/清零 (fetchClanLeaderboard)
 */
function fetchClanLeaderboard(options = {}, callback) {
  const { realSeasonId = 'RS_2026_Q4_HANWEI', myFamily = null, forceRefresh = false } = options;

  if (!forceRefresh && _cachedCloudLeaderboards[realSeasonId]) {
    const merged = _aggregateWithLocalFamily(_cachedCloudLeaderboards[realSeasonId], myFamily);
    if (typeof callback === 'function') callback(null, merged);
    return merged;
  }

  if (typeof wx !== 'undefined' && wx.cloud && typeof wx.cloud.callFunction === 'function') {
    wx.cloud.callFunction({
      name: 'clanLeaderboard',
      data: {
        action: 'getLeaderboard',
        seasonId: realSeasonId
      }
    }).then(res => {
      if (res && res.result && Array.isArray(res.result.clans)) {
        _cachedCloudLeaderboards[realSeasonId] = res.result.clans;
        const merged = _aggregateWithLocalFamily(res.result.clans, myFamily);
        if (typeof callback === 'function') callback(null, merged);
      } else {
        const fallback = _aggregateWithLocalFamily(SEED_SAMPLE_CLANS, myFamily);
        if (typeof callback === 'function') callback(null, fallback);
      }
    }).catch(err => {
      console.warn('[ClanCloudService] 拉取全服榜单异常，回退本地榜单:', err);
      const fallback = _aggregateWithLocalFamily(SEED_SAMPLE_CLANS, myFamily);
      if (typeof callback === 'function') callback(err, fallback);
    });
    return _aggregateWithLocalFamily(SEED_SAMPLE_CLANS, myFamily);
  }

  const localRes = _aggregateWithLocalFamily(SEED_SAMPLE_CLANS, myFamily);
  if (typeof callback === 'function') callback(null, localRes);
  return localRes;
}

function _aggregateWithLocalFamily(list, myFamily) {
  const cloned = [...list];
  if (myFamily && myFamily.familyId) {
    const idx = cloned.findIndex(c => c.familyId === myFamily.familyId);
    if (idx >= 0) {
      cloned[idx] = myFamily;
    } else {
      cloned.unshift(myFamily);
    }
  }
  return cloned;
}

/**
 * 《风华年代：沧海沉浮录》排行榜鉴权与微信开放数据域同步服务 (Leaderboard Sync Service)
 * 职责：终局双向签名计算、云函数防作弊验算门禁、写入 setUserCloudStorage 与 SharedCanvas 消息调度
 */

const SERVER_SALT_LIFESIM = 'LIFESIM_2026_ANTITAMPER_KEY_#8899';

function generateScoreChecksum(score, year, age, rep, clean, runSeed) {
  const str = `${score}:${year}:${age}:${rep}:${clean}:${runSeed}:${SERVER_SALT_LIFESIM}`;
  let hash = 0x811c9dc5;
  for (let i = 0; i < str.length; i++) {
    hash ^= str.charCodeAt(i);
    hash = Math.imul(hash, 0x01000193);
  }
  return (hash >>> 0).toString(16);
}

function createLeaderboardSync(deps = {}) {
  const {
    getState = () => ({}),
    evaluateEndgameArchetype = () => ({ grade: 'B', title: '平凡一生' }),
    sanitizeUserInput = (s) => s,
    reportMilestone = () => {},
    playSoundEffect = () => {},
    requestRender = () => {},
    logDebugError = () => {},
    getConfig = () => null
  } = deps;

  let _lastLeaderboardSyncTs = 0;
  const LEADERBOARD_SYNC_MIN_INTERVAL_MS = 5000;

  function syncScoreToLeaderboard() {
    try {
      const now = Date.now();
      if (now - _lastLeaderboardSyncTs < LEADERBOARD_SYNC_MIN_INTERVAL_MS) {
        return;
      }
      _lastLeaderboardSyncTs = now;

      const state = getState();
      const arch = state.endgameArchetype || evaluateEndgameArchetype(state);
      const dynCfg = (typeof getConfig === 'function' && getConfig()) || (typeof LIFESIM_DYNAMIC_CONFIG !== 'undefined' ? LIFESIM_DYNAMIC_CONFIG : null);
      const rew = (dynCfg && dynCfg.rewards) || {};
      const sScore = (rew.endgameSGradeBaseScore !== undefined) ? rew.endgameSGradeBaseScore : 1000;
      const aScore = (rew.endgameAGradeBaseScore !== undefined) ? rew.endgameAGradeBaseScore : 750;
      const bScore = (rew.endgameBGradeBaseScore !== undefined) ? rew.endgameBGradeBaseScore : 500;
      const cScore = (rew.endgameCGradeBaseScore !== undefined) ? rew.endgameCGradeBaseScore : 250;
      const gradeScore = arch.grade === 'S' ? sScore : (arch.grade === 'A' ? aScore : (arch.grade === 'B' ? bScore : cScore));
      // 排除 B 组广告额外赠予的 ¥15,000 与 +5 声望，维护全平台排行榜基准公平
      const effectiveRep = Math.max(0, (state.rep || 0) - (state.heritageBonusRep || 0));
      const effectiveCapital = Math.max(0, (state.capital || 0) - (state.heritageBonusCapital || 0));
      const rawScore = Math.round(gradeScore + effectiveRep * 5 + (state.clean || 0) * 3 + Math.min(500, Math.floor(effectiveCapital / 5000)));
      const delegationPenalty = (state.delegatedDecisionsCount && state.delegatedDecisionsCount > 0) ? 0.75 : 1.0;
      const totalScore = Math.round(rawScore * delegationPenalty);

      const isHeadless = Boolean(state._isHeadlessSimulation || (typeof global !== 'undefined' && global.__HEADLESS_SIM__));

      if (!Array.isArray(state.leaderboardHistory)) state.leaderboardHistory = [];
      const cleanedName = sanitizeUserInput(state.name || '时代弄潮儿');
      const safeActorName = (!cleanedName || cleanedName.includes('***')) ? '时代弄潮儿' : cleanedName;
      state.leaderboardHistory.push({
        name: safeActorName,
        score: totalScore,
        grade: String(arch.grade || 'B'),
        title: String(arch.title || ''),
        age: state.age || 78,
        year: state.year || 2050,
        track: state.track || 'gov',
        bucket: state.abTestBucket || 'A_standard',
        ts: Date.now()
      });
      state.leaderboardHistory.sort((a, b) => (b.score || 0) - (a.score || 0));
      if (state.leaderboardHistory.length > 10) {
        state.leaderboardHistory = state.leaderboardHistory.slice(0, 10);
      }

      if (isHeadless) return;

      const secureChecksum = generateScoreChecksum(
        totalScore,
        state.year || 2050,
        state.age || 78,
        state.rep || 50,
        state.clean || 50,
        state.runSeed || 0
      );

      const writeVerifiedCloudStorage = (finalScore, finalGrade, finalTitle) => {
        if (typeof wx !== 'undefined' && typeof wx.setUserCloudStorage === 'function') {
          wx.setUserCloudStorage({
            KVDataList: [
              { key: 'era_score', value: String(finalScore) },
              { key: 'era_grade', value: String(finalGrade || 'B') },
              { key: 'era_title', value: String(finalTitle || '') },
              { key: 'era_bucket', value: String(state.abTestBucket || 'A_standard') }
            ],
            success: () => {
              if (typeof wx.getOpenDataContext === 'function') {
                const openDataContext = wx.getOpenDataContext();
                if (openDataContext && typeof openDataContext.postMessage === 'function') {
                  openDataContext.postMessage({
                    type: 'UPDATE_SCORE',
                    score: finalScore,
                    grade: finalGrade,
                    title: finalTitle
                  });
                }
              }
            }
          });
        }
        reportMilestone('leaderboard_synced', { score: finalScore, grade: finalGrade });
      };

      if (typeof wx !== 'undefined' && wx.cloud && typeof wx.cloud.callFunction === 'function') {
        try {
          wx.cloud.callFunction({
            name: 'verifyAndSubmitScore',
            data: {
              score: totalScore,
              rawScore: rawScore,
              grade: arch.grade,
              title: arch.title,
              runSeed: state.runSeed,
              year: state.year,
              age: state.age,
              rep: state.rep,
              clean: state.clean,
              capital: state.capital,
              delegatedDecisionsCount: state.delegatedDecisionsCount || 0,
              checksum: secureChecksum,
              submitTime: now
            }
          }).then((res) => {
            if (res && res.result && res.result.verified) {
              writeVerifiedCloudStorage(res.result.score || totalScore, arch.grade, arch.title);
            } else {
              console.warn('[Leaderboard] 云函数校验拦截，拒绝写入开放数据域:', res && res.result);
            }
          }).catch((err) => {
            console.warn('[Leaderboard] 云函数调用未响应，保持本地成绩:', err);
          });
        } catch (e) {
          logDebugError('syncScoreToLeaderboard.cloud', e);
        }
      } else {
        if (typeof wx !== 'undefined' && typeof wx.getSystemInfoSync === 'function') {
          try {
            const sys = wx.getSystemInfoSync();
            if (sys && sys.platform === 'devtools') {
              writeVerifiedCloudStorage(totalScore, arch.grade, arch.title);
            }
          } catch (_) {}
        }
      }
    } catch (err) {
      logDebugError('syncScoreToLeaderboard', err);
    }
  }

  function openLeaderboardModal(screenWidth) {
    const state = getState();
    state._leaderboardReturnModal = state.activeModal === 'endgame' ? 'endgame' : null;
    state.activeModal = 'leaderboard';
    try {
      if (typeof wx !== 'undefined' && typeof wx.getOpenDataContext === 'function') {
        const openDataContext = wx.getOpenDataContext();
        if (openDataContext && typeof openDataContext.postMessage === 'function') {
          openDataContext.postMessage({ type: 'SHOW_LEADERBOARD', width: screenWidth - 48, height: 220 });
        }
      }
    } catch (err) {
      logDebugError('openLeaderboardModal', err);
    }
    playSoundEffect('paper');
    requestRender();
  }

  return {
    syncScoreToLeaderboard,
    openLeaderboardModal,
    generateScoreChecksum
  };
}

## 四、 线上内容热更新引擎与安全防篡改体系

微信小游戏严禁动态下发并执行网络 `.js` 脚本。本项目设计了纯 JSON 增量补丁热更新引擎（`contentUpdater.js`）：
1. **权威信道下发**：
   * 优先通过 `clanLeaderboard.getContentManifest` 云函数原生下发，自带微信平台环境鉴权标识；
   * 若走 HTTP/CDN 备用信道，强制验证数字签名，**坚决拒绝客户端源码内置公开默认盐（`HOTUPDATE_DEFAULT_SALT`）**，必须配置私有安全密钥或走云端鉴权。
2. **原型污染防御与数值白名单裁剪**：
   * 递归清理 `__proto__`、`constructor`，杜绝任何外部注入风险；
   * 建立数值限幅白名单：
     * 金钱与奖金项（`Money`、`capital`、`gold`、`Fund`、`Bonus`、`Cash`、`Salary`）：限幅至 `[-500000, 500000]`；
     * 积分与考核基准分（`Score`、`Threshold`、`BaseScore`，如 `promotionBonus: 22000`, `endgameSGradeBaseScore: 1000`）：限幅至 `[0, 50000]`；
     * 基础单次属性微调项：限幅至 `[-30, 30]`；
     * 岁月前沿线：放开至 `[2026, 3100]` 年。

## 五、 最小赛季内容包 (MSCP) 与 7 周产线倒排排期

为了保证每个 200 年纪元赛季的高品质持续交付，制定了 **MSCP (Minimum Season Content Pack - 最小赛季内容包)** 工业级准入标准：

### 1. MSCP 交付规格
* **官制阶梯**：至少 5 条主线阶梯（文臣/学士/商贾/武弁/隐逸），每条梯级 5 阶（共 25 阶）；
* **纪元名宿**：6 ～ 10 位历史名仕立绘、台词文风与专属好感羁绊；
* **重大关隘事件**：4 ～ 12 个历史关键抉择事件（含检定要求、代际影响、谥号盖棺标记）；
* **传世信物与家训**：2 ～ 3 件纪元特色传世重宝信物（具备长线专属被动加成）；
* **纪元基础池**：出生地郡望池 ($\ge 6$)、父辈职业池 ($\ge 5$)、抓周物池 ($\ge 6$)。

### 2. 7 周倒排排期标准 (T-7 到 T-0)
* **T-7 周**：史实考据与断代立项，确定世家谱系脉络与 5 条官制阶梯初稿；
* **T-6 周**：数值框架搭建，平衡俸禄产出系数与门第声望阈值；
* **T-5 周**：剧情编撰，设计 4~12 个历史关隘大案与选项分流；
* **T-4 周**：名宿通讯录研制，完成 6~10 位历史名仕交互文案与增益链；
* **T-3 周**：传世信物与家训策划，设计长线专属宗族加成；
* **T-2 周**：无头自动化仿真跑测，验证代际流转零死循环与数值收敛；
* **T-1 周**：灰度体验、双模合规核查与数字签名封包；
* **T-0 周**：开季正式上线，全服平滑无感升级。

> **未研制内容包的赛季（如第3赛季隋唐草创，定档 2027-01-23）**：结算弹窗自动锁死启程按钮，显示“⏳ 敬请期待（定档 2027-01-23）”，引导玩家进入休赛期四大经略（多房分支再起、宗祠家产深耕、世家秦晋联姻、风云青史冲榜）。

## 八、 移动端性能极限压榨、内存防漏与架构健壮性专项治理

针对小游戏运行中出现的“静止时整屏重绘发热”、“装饰性静态层无离屏缓存”、“切后台定时器泄漏”及“长局存档无限增长”等核心痛点，在 `wechat-minigame` 与 `douyin-minigame` 双端落地四重深度重构：

### 1. 静止重绘根治：动态分级帧频调步与粒子遮挡冻结
- **根因分析**：常驻 40 颗丁达尔金粉微粒（`ambientMotes`）在每一帧无条件执行三角函数浮点计算与坐标更新，致使无论用户是否处于静态阅读弹窗或空闲静止状态，引擎均在以 ~30FPS 不间断进行整屏清屏与重绘，造成发热与电量消耗。
- **粒子遮挡冻结**：在 `animationTick()` 中引入模态弹窗遮挡检测 `modalAnimState.progress >= 0.7` 以及低端机模式判断；当弹窗展开或降级模式激活时，彻底停止微粒坐标位移物理更新，消除 CPU 空转。
- **分级自适应帧频**：
  - **高动态交互态（60FPS）**：手指触控操作、惯性弹性滚动、Tab 滑入、骰子旋转、墨水喷溅、金币粒子或 Toast 浮现时，渲染间隔 `targetInterval = 0`，维持丝滑 60FPS。
  - **空闲静止态（15FPS）**：无任何动态操作时，帧率主动降至 15FPS（`targetInterval = 66ms`），相比原先 30FPS（32ms）直接减半重绘开销。
  - **阅读弹窗态（10FPS）**：模态弹窗完全展开且静态阅读时，帧率降至 10FPS（`targetInterval = 100ms`），彻底遏制发热与功耗。

### 2. 静态装饰层离屏画布缓存加速（Offscreen Canvas Texture Caching）
- **离屏缓冲池封装**：封装多端兼容的 `createOffscreenCanvasBuffer(w, h)`，优先适配 `wx.createOffscreenCanvas` / `tt.createOffscreenCanvas`（2D），优雅降级至 `wx.createCanvas` / `tt.createCanvas` 及浏览器 Canvas。
- **宋锦缠枝云纹画框（`drawCloudFiligreeBorders`）**：引入 LRU Map 离屏纹理缓存，将四角祥云回字纹、金线描边等多次 `beginPath` / `moveTo` / `lineTo` / `stroke` 调用预渲染为离屏纹理，每帧直接通过 `ctx.drawImage` 单次贴图，消除高频 CPU 路径栅格化开销。
- **中式宣纸装裱卷轴（`drawSilkScrollFrame`）**：将底层宣纸衬底、顶部与底部实木轴杆、双侧鎏金轴头及祥云角花整体打包为离屏图像缓存，大幅降低重绘耗时。
- **顶部天幕微光渐变（`drawTopBar`）**：复用并缓存 `_cachedTopGrad` 线性渐变对象，避免每帧重复创建与垃圾回收（0 GC）。

### 3. 定时器全生命周期闭环与切后台安全守护
- **巡航定时器安全闭环**：实现全局 `stopAutoAdvance()`，统一销毁 `state.autoTimer` 并复位 `state.isPlaying = false`；在遇到终局（`endgame`）、手动暂停以及长按岁月快进（`rapidAdvanceTimer`）冲突时严格清除。
- **多端后台挂起监听**：全面对齐并兼容 `tt.onHide` / `wx.onHide`、`tt.onShow` / `wx.onShow` 及音频中断事件。在切后台瞬间，强制执行：
  - `stopBGM()` 停止音频与音频调度定时器；
  - `stopAutoAdvance()` 销毁自动巡航定时器；
  - `stopRapidAdvance()` 销毁长按快进定时器；
  - `saveGameState(true)` 同步落地静止存档，避免后台被系统强杀丢档。

### 4. 存档体积上限与大局历史数据截断（Prevent Save Bloat）
- **编年与画卷属性对齐**：修复 `addLifeTrajectoryPoint` 中历史变量名解耦偏差（`state.chronicle` 修正为规范属性 `state.chronicleHistory`），并将 `lifeWaveHistory` 与 `chronicleHistory` 严格约束为最多 120 节点。

  彻底消除百岁长局、多代传承时存档体积无限膨胀导致的同步写盘卡顿与内存泄漏隐患。

## 3. 核心设计（一页纸）

### 3.1 一句话玩法
**在时代浪潮里，用每回合有限的精力，把一生押在几次关键抉择上。**

### 3.2 一局结构与时间预算（5 分钟版）
| 环节 | 现状 | 新设计 | 预算 |
|---|---|---|---|
| 开局（出身、天赋） | 创角流程较长 | 只选一次关键项，其余随机 | ~30 s |
| 童年（0–17 岁） | 10 个固定事件 | **一次性跑完，保留 1 次分流选择**（抓周/少年分流），其余自动 | ~20 s |
| 成年（18–78 岁） | 逐年推进，每年可能弹窗 | **12–13 个回合，每回合 = 5 年** | ~3 min 30 s |
| 结局与传承 | 有 | 结局卡 + 一次传承选择 | ~40 s |

### 3.3 回合流程
1. **回合开始**：展示时代节点（如"1992 南巡"），玩家选择**本回合方针**（进取 / 清廉 / 人脉 / 养生）。
2. **精力分配**：本回合有 3 点精力（不累积），分配给四类行动（建议，见 §3.4）。
3. **回合推进**：引擎连续推进最多 5 年；遇到 **A 类事件**（见 §3.5）停下等玩家；其余自动处理。
4. **回合小结卡**：一张卡汇总本回合被代办/自动处理的事件及其结果（一句话回响）。
5. 进入下一回合，直到结局。

> 预期每回合约 15 秒：看节点（3s）→ 选方针（2s）→ 分配精力（4s）→ 一次抉择（5s）→ 看小结（1–2s）。

### 3.4 资源与稀缺（建议，数值待校准）
- **精力**：每回合 3 点，不累积。分配类别（初版建议）：
  - **事业**：为本回合的 A 类检定提供加成（可复用现有 `insightBuff`，+30% 的机制已存在）。
  - **修身**：恢复健康、抬高健康上限折损的恢复速度。
  - **家庭**：推进家族线（子嗣、传家宝）。
  - **人脉**：提高"幕僚代办"的质量（见 §3.6）。
- **健康是时钟**：高强度事业加速消耗，防止"全部拉满"。
- **声望 / 清廉 / 资金有兑换损耗**：不能同时无成本拉满（具体规则请执行 Agent 与产品商定；起点见 P3）。
- 现有 AP 系统（占卜、NPC 对话、家事、一线抉择、养生等）**保留并折算到回合精力里**，不要另起一套。

### 3.5 事件分级（A / B / C）
| 级别 | 含义 | 内容 | 处理 |
|---|---|---|---|
| **A 必须亲自** | 叙事主线、转折 | 历史大事件（**每回合最多 1 个**）、危机（审计/债务/健康/黑天鹅）、晋升、终身大考、转赛道、重大工程 | 停下等玩家 |
| **B 填充** | 保证差异的变量内容 | 赛道专属、程序化遭遇、市井奇遇、节气 | **本回合没有 A 时出 1 个**；设保底槽位（见 P4） |
| **C 自动** | 纯确认/阅读 | 考评报告、史官赞、书简、荣典、因果回响、夫妻手札、世纪回望等 | 合并进回合小结，不弹窗 |

- **未知事件默认 B，未知弹窗默认 C**，并在开发版日志里打印，便于补表。
- 分级表放在**独立的数据表** `EVENT_TIER_TABLE`，不要散落在逻辑里。

### 3.6 幕僚代办规则
- 超出本回合 A 类上限的事件**不丢弃**，由"幕僚"自动结算，**效果照常生效**，并写入回合小结。
- **不要直接用 `resolveCurrentModalByPolicy('strategic')` 当代办策略** [推断]：它是 QA 用的自动评测函数，评估公式能看到全部数值，比真人强，会把难度压没。建议新建 `delegatePolicy`：
  - 基础：选通过率较高的稳妥项；
  - 噪声：加入随机扰动；
  - 质量：随"人脉"精力和情商提高。
- **陷阱（已验证）**：`tickOneYear` 开头有"未决弹窗兜底惩罚"（约 18165–18186 行：`unresolvedCrisis`、`unresolvedDecisionEvent` 会扣声望/清廉/资金）。代办结算**必须清掉这两个标记**，否则下一年会被误罚。`executeDiceChoice` 和三个危机结算函数已会清，但新写的代办路径要逐个核对。

### 3.7 目标、结局与传承
- **开局选一个人生志向**（沿用 `LIFE_AMBITION_TARGETS`），HUD 全程显示进度。
- **结局按终局状态判定**，不再"曾经达成就永久为真"；可同时展示"曾达成"。
- **传承**：失败局也有保底（建议 +1 点和一条结局图鉴）；点数可**兑换永久解锁**（新出身、新天赋、新赛道、改写一个历史节点）。
- **每日挑战**（后置）：同一个种子，全员同一个人生，按结局评分排名。

## 4. 工作分解

> 规模：S ≈ 半天内，M ≈ 1–3 天，L ≈ 3 天以上（粗略估计，仅供排期参考）。
> 依赖：P0 → P1 → P2 → P3 → P4 → P5；P7 可并行；P6 在 P5 之后。

### P0 基线与验收工具（S）
**目标**：改动前先固化度量，改动中持续回归。
**做法**
1. 运行 `node balance_sim.js 1790845892887_game.js 2 all --json baseline_report.json`，确认与本文基线一致。
2. 在游戏里补充**真机计时埋点**，用来校准"12 秒/次交互"的假设：
   - `turn_started` / `turn_completed`（含 `duration_ms`、`player_actions`、`delegated_count`）；
   - `session_length`；
   - `ap_spent`（按类别）。
   - 复用现有 `reportGameMilestone`（现有事件见附录 A）。
3. 建立 `CHANGELOG_BALANCE.md`：每次改动记录"改了什么 / 指标前后对比"。

**验收**：基线可复现；埋点在开发版日志可见。

### P1 回合化与降噪（L，**最高优先级**）
**目标**：交互数 ≤ 20，估算时长 ≤ 5.5 分钟，且变量内容能出场。
**改动点**
- 新增 `TURN_MODE` 开关；新增 `runTurn()`（脚本已约定接口，见 `balance_sim.js` 头部注释）。
- 新增 `EVENT_TIER_TABLE`（见附录 B 的初版分级）。
- 新增回合小结弹窗 `activeModal = 'turn_digest'`。
- 童年：`tickChildhoodYear`（约 14558 行，由 `tickOneYear` 在 `age < 18` 时调用）改为**一次性跑完 0–17 岁**，只保留 1 次分流。
- 寿命：集中到一个 `getLifespanCeiling()`，硬上限 85（详见 P3）。
- 自动巡航：`togglePlayPause`（约 20762 行，`1800 / playSpeed` ms/年）改为"自动进入下一回合"，移除/重做倍速。

**`runTurn` 参考骨架（伪代码，需按实际弹窗队列调整）**
```js
function runTurn() {
  if (state.activeModal || state.pendingPhoneCrisis) return;
  if (!state.turn || state.turn.yearsLeft <= 0) beginTurn();   // 设定 index/startAge/yearsLeft=5/aBeats=0/digest=[]
  while (state.turn.yearsLeft > 0) {
    tickOneYear();                         // 年度引擎，不改动
    state.turn.yearsLeft--;
    while (state.activeModal) {            // 逐个处理本年弹出的弹窗
      if (state.activeModal === 'endgame') return;
      const tier = classify(state.activeModal, state.activeDecisionEvent);
      if (tier === 'A' && state.turn.aBeats < A_CAP) {
        state.turn.aBeats++;
        return;                            // 停下，等玩家；玩家结算后 UI 再调用 runTurn() 继续
      }
      delegate(tier);                      // 代办/自动结算，写入 state.turn.digest，并清未决标记
    }
    flushNextQueuedModal();                // 复用现有队列
  }
  endTurn();                               // 弹出 turn_digest
}
```
**注意**
- 玩家结算停下的弹窗后，需要能**继续本回合剩余年份**（因此要存 `yearsLeft`，并写进存档）。
- `enqueueModalAction` / `flushNextQueuedModal`（约 12689/12700 行）是现有弹窗队列，**复用，不要另起**。
- 先用离线日志验证分级：我已用 24 局日志测算（"每回合 1 个 A、童年 1 次分流、寿命 ≤78–80"→ 13–14 次选择、约 4.2–4.3 分钟）；落地后以模拟器实测为准。

**验收**：见 §6 的 P1 项；报错为 0；`skip` 仍 0% S/A。
**风险**：见 §7。

### P2 稀缺与回合方针（M）
**目标**：让"每回合怎么分配"成为真正的决策。
**做法**
1. **暴露年度策略入口**：把 `state.strategy` 改为"回合方针"，在回合开始时让玩家选；原年度结算效果（约 17932–17946 行）改为每回合一次。
2. **精力分配 UI**：回合开始的小面板（事业/修身/家庭/人脉）。
3. **折算现有 AP**：确认成年阶段的 AP 刷新位置 [待确认]（童年刷新在约 14564 行，成年处未找到）；`unlockTalent('decisive')` 把上限设为 4（约 8262 行）要保留语义（"决断"天赋 = 每回合 +1 精力）。
4. 事业精力接到 `executeDiceChoice` 的成功率修正（现有 `insightBuff`、`hasTalent('breakthrough')` 可作参考实现）。
5. 补上程序化事件的 `statReq`（见 P4）。

**验收**：精力总预算 ≈ 每局 40 点；`skip` 仍 0%；各策略 S/A 占比开始下降（本阶段不要求达标）。

### P3 曲线与后果（M）
**目标**：50 岁之后仍有成长与风险；危机不再是重置键。
**做法（数值均为起点）**
1. **成长递减**：属性收益按当前值衰减，例如 `实际收益 = 原收益 × max(0.25, (100 − 当前值) / 60)`。
2. **位高责重**：声望 ≥ 80 时，审查/舆情类事件概率上升、失败代价加重。
3. **危机"带疤痕"**：
   - 审计：不再 `max(清廉,72)`；保留真实清廉下降，并加 3 个回合的"在案"标记（晋升判定扣分）。
   - 债务：不再直接 `capital = 28000`；保留一部分欠款（例如按比例折算）。
   - 健康：不再直接恢复到 68；留下健康上限永久折损（例如 −5）。
4. **寿命硬上限**：在 `getLifespanCeiling()` 里统一 `min(…, 85)`；原有 `+4~+8` 的加成改为"健康衰减更慢"。涉及 `lifespanCeiling` 的所有写入点（见附录 A）。
5. **属性写入收口**：全部通过 `addStat`；修复 `state.rep = (state.rep || 50) + …` 里"声望为 0 被当作 50"的问题。

**验收**：strategic 的 50 岁声望 ≤ 80；平均终龄 70–80；strategic S/A 降到 40–50%，random 10–15%，skip 0%。

### P4 变量内容与赛道差异（L）
**目标**：每局不一样；每条赛道都有自己的故事。
**做法**
1. **B 类保底槽位**：回合里没有 A 时，按"赛道专属 > 程序化 > 市井奇遇"的顺序出 1 个 B；并要求每局至少 N 个来自赛道专属（建议 N≥3，待校准）。
2. **补赛道内容**：程序化年度事件池（`PROCEDURAL_ANNUAL_ENCOUNTER_POOL`，约 14356 行）里 **med / law / military 无专属事件**，art、trade、education、grassroots 很少；建议每条赛道至少 15 个专属事件。
3. **补 `statReq`**：程序化事件现有 `statReq` 约 60 处，仍有缺口；让属性培养真正影响成功率。
4. **历史事件去重**：核对附录 B 里同年的 `arc_*` 与 `epic_*` 是否重复（S11），重复则合并。
5. **赛道转换**：`state.track = 'trade'`（约 20243 行）改为写入 `state.subTrack`，主赛道不变，S 档判定和 HUD 同时展示。
6. **美术、法律 S 档结局**：补到 `ENDGAME_ARCHETYPES_16`（见 P5）。

**验收**：固定剧本占比 ≤ 50%；变量决策 ≥ 8/局；120 局内各赛道的赛道专属事件出现率 ≥ 设定值。

### P5 目标、结局、传承（M）
**目标**：目标驱动全程；结局可收集；传承有出口。
**做法**

## 6. 验收与回归

### 6.1 命令
```bash
# 基线（改动前跑一次并保存）
node balance_sim.js 1790845892887_game.js 2 all --json baseline_report.json

# 每个阶段完成后
node balance_sim.js <新的 game.js> 2 all --json after_Px.json
```
- 参数：`<game.js> [每赛道局数] [skip|random|strategic|all] [--json 输出]`。
- 脚本不改动原文件；在临时副本末尾追加导出。
- 脚本会优先调用 `runTurn()`（如果存在）；估算时长里的 12 秒/次是假设，**请用 P0 的真机埋点校准后改脚本常量**。

### 6.2 各阶段的通过标准
| 阶段 | 必须满足 |
|---|---|
| P0 | 基线可复现（与 §1.2 一致）；埋点可见 |
| P1 | 交互/局 ≤ 20；估算时长 ≤ 5.5 分钟；`skip` S/A = 0%；报错 = 0；`unresolved*` 标记无误罚（检查日志） |
| P2 | `skip` S/A = 0%；精力预算和方针入口生效；报错 = 0 |
| P3 | strategic 50 岁声望 ≤ 80；平均终龄 70–80；strategic S/A 40–50%，random 10–15% |
| P4 | 固定剧本占比 ≤ 50%；变量决策 ≥ 8/局；med/law/military 有专属事件 |
| P5 | 120 局内结局 ≥ 12/16；各赛道 S 档可达；传承点有出口 |
| 全部 | 报错 0；`Math.random` 仍为 0；存档迁移通过（旧存档能读） |

### 6.3 模拟器的局限（请知悉）
- `strategic` 用的是游戏自带的评测策略，**比真人强**，它的 S/A 占比是上限，不是玩家平均水平。
- 样本小（每策略 24 局），只能看趋势；正式验收请加大到每赛道 ≥ 20 局。
- 事件分级、固定剧本统计依赖决策 ID；新增事件要保证 ID 稳定、可读。
- 模拟器只处理弹窗的"确认/选择"，**不衡量手感、动效、阅读体验**。

## 7. 风险与回退
| 风险 | 说明 | 缓解 |
|---|---|---|
| `runTurn` 包不住所有弹窗路径 | 部分弹窗在点击回调里才产生后续弹窗 | 先在模拟器里跑通 12 赛道 × 多局，再上真机；`TURN_MODE` 随时关闭 |
| 代办导致玩家失去控制感 | 一半历史大事件不再亲自决策 | 小结卡写清楚结果；人脉精力可提升代办质量；做 A/B 对比留存 |
| 代办策略过强/过弱 | 直接复用评测策略会压没难度 | 使用独立 `delegatePolicy`（§3.6） |
| 回合化后叙事节奏变化 | 5 年一回合，历史节点不再逐年出现 | 回合标题用"时代节点"，小结卡保留年份与事件名 |
| 数值联动 | P3 成长递减可能让所有赛道都"打不满" | 每步跑模拟器，保留 `skip=0%` 和 `strategic≈40–50%` 的双锚点 |
| 存档迁移 | 回合状态写进存档 | 写迁移与回归用例；保留旧存档读取路径 |
| 工程债 | 单文件继续增长 | P1 之后把事件数据外置、拆分模块 |
